Exemple

Simple par conception

Une API expressive qui se lit comme du français naturel.

public/index.php 
// Démarrer l'application
$app = new Application();

// Routes avec groupes et middlewares
$app->routeur->groupe([
    'prefixe'     => '/admin',
    'middlewares' => [AuthMiddleware::class],
], function ($r) {
    $r->get('/tableau', [AdminControleur::class, 'tableau']);
    $r->delete('/articles/{id}', [AdminControleur::class, 'supprimer']);
});

// Validation + ORM en une fois
$req->valider([
    'email'        => 'requis|email|unique:utilisateurs',
    'mot_de_passe' => 'requis|min:8|confirme',
]);

Article::paginer($page, 15); // → Paginateur

$app->demarrer();
